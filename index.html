// Geração do PDF (Vítrea e Cligest) window.gerarGuia = async (id, instituicao) => { try { showToast(`A gerar guia para a ${instituicao}...`); // 1. Ir buscar os dados do paciente e a assinatura ao Firebase const pSnap = await getDoc(doc(db, "pedidos", id)); const sSnap = await getDoc(doc(db, "configuracoes", "direcao")); if (!pSnap.exists()) return; const p = pSnap.data(); const assinaturaB64 = sSnap.exists() ? sSnap.data().assinatura_base64 : ""; let templateId = ""; let filename = ""; // 2. Preencher o molde correto consoante a instituição if (instituicao === "Vítrea") { document.getElementById('pdf-template-vitrea').style.display = 'block'; document.getElementById('pdf-template-cligest').style.display = 'none'; document.getElementById('p-nome').textContent = p.nome; document.getElementById('p-contacto').textContent = p.contacto; document.getElementById('p-dataReal').textContent = p.dataExame; document.getElementById('p-exame').textContent = p.exame; document.getElementById('pdf-assinatura').src = assinaturaB64; templateId = 'pdf-template-vitrea'; filename = `Guia_Vitrea_${p.nome.replace(/\s+/g, '_')}.pdf`; } else if (instituicao === "Cligest") { document.getElementById('pdf-template-vitrea').style.display = 'none'; document.getElementById('pdf-template-cligest').style.display = 'block'; document.getElementById('c-nome').textContent = p.nome; // Inverte a data de nascimento de YYYY-MM-DD para DD/MM/YYYY document.getElementById('c-nascimento').textContent = p.nascimento ? p.nascimento.split('-').reverse().join('/') : ''; document.getElementById('c-contacto').textContent = p.contacto; document.getElementById('c-medico').textContent = p.medico; document.getElementById('c-exame').textContent = p.exame; document.getElementById('c-dataReal').textContent = p.dataExame ? p.dataExame.split('-').reverse().join('/') : ''; document.getElementById('pdf-assinatura-cligest').src = assinaturaB64; templateId = 'pdf-template-cligest'; filename = `Guia_Cligest_${p.nome.replace(/\s+/g, '_')}.pdf`; } else { alert("Instituição desconhecida."); return; } // 3. Gerar o PDF e forçar o download const opt = { margin: 0, filename: filename, html2canvas: { scale: 2 }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait' } }; await html2pdf().set(opt).from(document.getElementById(templateId)).save(); showToast("PDF descarregado com sucesso!"); } catch (e) { console.error(e); alert("Erro ao gerar PDF."); } };
